    <p class="section-text">

<section class="services-section">

    <h2>
      Beratung &<br>
      Konzepte.
    </h2>

    <p class="section-text">
      Wir beraten Bauherren bei technischen Fragestellungen und entwickeln
      geeignete Lösungen für Neubauten, Umbauten und Sanierungen.

    </p>
  </div>
</section>

<section class="services-section">
  <div class="section-inner">

    <h2>
      Berechnung &<br>
      Dimensionierung.
    </h2>

    <p class="section-text">

    </p>
  </div>
</section>
